local function getVehicleModel()
    if humanoid and humanoid.SeatPart then
        local current = humanoid.SeatPart
        while current and current ~= workspace do
            if current:IsA("Model") then
                return current
            end
            current = current.Parent
        end
    end
    return nil
end

local function smoothMoveVehicle(vehicle, destinationCFrame, duration)
    local startCF = vehicle:GetPivot()
    local startUnderground = startCF - Vector3.new(0, 25, 0)
    local targetUnderground = destinationCFrame - Vector3.new(0, 25, 0)
    
    for _, part in ipairs(vehicle:GetDescendants()) do
        if part:IsA("BasePart") then
            part.AssemblyLinearVelocity = Vector3.zero
            part.AssemblyAngularVelocity = Vector3.zero
        end
    end

    local function lerpMove(fromCF, toCF, moveTime)
        local elapsed = 0
        while elapsed < moveTime do
            local delta = RunService.Heartbeat:Wait()
            elapsed = elapsed + delta
            local alpha = math.clamp(elapsed / moveTime, 0, 1)
            
            vehicle:PivotTo(fromCF:Lerp(toCF, alpha))
            if vehicle.PrimaryPart then
                vehicle.PrimaryPart.AssemblyLinearVelocity = Vector3.zero
            end
        end
    end

    lerpMove(startCF, startUnderground, 1)
    lerpMove(startUnderground, targetUnderground, duration)
    lerpMove(targetUnderground, destinationCFrame, 1.5)
    vehicle:PivotTo(destinationCFrame)
end

local function triggerPrompt(prompt)
    if prompt and prompt:IsA("ProximityPrompt") then
        prompt.HoldDuration = 0
        prompt.MaxActivationDistance = 9999
        fireproximityprompt(prompt)
    end
end

local function getItemPrompt(itemName)
    local smugglingItems = workspace:FindFirstChild("Smuggling") and workspace.Smuggling:FindFirstChild("Items")
    if not smugglingItems then return nil end

    for _, item in ipairs(smugglingItems:GetChildren()) do
        if item.Name == itemName or string.find(item.Name, itemName) then
            local main = item:FindFirstChild("Main")
            if main then
                local promptAtt = main:FindFirstChild("PromptAtt")
                if promptAtt then
                    local prompt = promptAtt:FindFirstChild("SmugglePurchasePrompt")
                    if prompt and prompt:IsA("ProximityPrompt") and prompt.Enabled then
                        return prompt
                    end
                end
            end
        end
    end
    return nil
end

local function hasBriefcase()
    local backpack = localPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") and string.find(item.Name:lower(), "briefcase") then
                return true
            end
        end
    end
    if character then
        for _, item in ipairs(character:GetChildren()) do
            if item:IsA("Tool") and string.find(item.Name:lower(), "briefcase") then
                return true
            end
        end
    end
    return false
end

local function startAutoFarmCycle()
    if isProcessing or not autoFarmActive then return end

    if not humanoid.SeatPart then
        print("[Auto-Farm]: Mashinaga o'tirishingiz kutilmoqda...")
        while not humanoid.SeatPart and autoFarmActive do
            task.wait(0.5)
        end
    end

    if not autoFarmActive then return end

    local vehicle = getVehicleModel()
    if not vehicle then return end

    isProcessing = true

    -- STEP 1: Fast Buy (0.6s cooldown)
    print("[1/4] Sotib olish joyiga borilmoqda...")
    smoothMoveVehicle(vehicle, buyCFrame, 1)
    task.wait(0.5)

    print("[1/4] Smuggle item'lar sotib olinmoqda (0.6s)...")
    for _, req in ipairs(buyList) do
        for i = 1, req.count do
            if not autoFarmActive then break end
            
            local prompt = getItemPrompt(req.name)
            if prompt then
                triggerPrompt(prompt)
                print("[BUY]: " .. req.name .. " (" .. i .. "/" .. req.count .. ")")
            end
            task.wait(0.6)
        end
    end

    if not autoFarmActive then isProcessing = false return end

    -- STEP 2: Move to Target CFrame & Continuous Prompt Fire
    print("[2/4] Sotish joyiga borilmoqda (45s)...")
    smoothMoveVehicle(vehicle, sellCFrame, 45)

    print("[2/4] Mashinadan tushildi. Sell promptlar uzluksiz urilmoqda...")
    local currentSeat = humanoid.SeatPart
    humanoid.Sit = false
    task.wait(0.5)

    while not hasBriefcase() and autoFarmActive do
        local sellFolder = workspace:FindFirstChild("Smuggling") and workspace.Smuggling:FindFirstChild("Sell")
        
        if sellFolder then
            for _, desc in ipairs(sellFolder:GetDescendants()) do
                if desc:IsA("ProximityPrompt") then
                    triggerPrompt(desc)
                end
            end
        else
            for _, desc in ipairs(workspace:GetDescendants()) do
                if desc:IsA("ProximityPrompt") and desc.Name:lower():find("sell") then
                    triggerPrompt(desc)
                end
            end
        end

        if humanoid then
            humanoid.Jump = true
        end
        task.wait(0.3)
    end

    if not autoFarmActive then isProcessing = false return end
    print("[SUCCESS]: Briefcase inventarda paydo bo'ldi! Washing'ga ketilmoqda...")

    if currentSeat and humanoid then
        currentSeat:Sit(humanoid)
    end
    task.wait(1.5)

    -- STEP 3: Money Wash
    print("[3/4] Washing machine joyiga ketilmoqda (7s)...")
    smoothMoveVehicle(vehicle, washCFrame, 7)

    print("[3/4] Mashinadan tushib pul yuvilmoqda...")
    currentSeat = humanoid.SeatPart
    humanoid.Sit = false
    task.wait(0.8)

    if humanoid then humanoid.Jump = true end
    
    local washPrompt = workspace:FindFirstChild("Smuggling") 
        and workspace.Smuggling:FindFirstChild("Laundering") 
        and workspace.Smuggling.Laundering:FindFirstChild("Prompt") 
        and workspace.Smuggling.Laundering.Prompt:FindFirstChild("SmuggleLaundryPrompt")

    if washPrompt then
        triggerPrompt(washPrompt)
        print("[LAUNDRY]: Pul muvaffaqiyatli yuvildi!")
    else
        for _, desc in ipairs(workspace:GetDescendants()) do
            if desc:IsA("ProximityPrompt") and (desc.Name:lower():find("laundry") or desc.Name:lower():find("wash")) then
                triggerPrompt(desc)
                print("[LAUNDRY-ALT]: Yuvish prompti urildi!")
                break
            end
        end
    end
    task.wait(2)

    if currentSeat and humanoid then
        currentSeat:Sit(humanoid)
    end
    task.wait(1.5)

    -- STEP 4: Recycling
    print("[4/4] Sotib olish joyiga qaytilmoqda...")
    smoothMoveVehicle(vehicle, buyCFrame, 1)

    print("[RECYCLING]: Sikl yakunlandi!")
    task.wait(1)
    isProcessing = false

    if autoFarmActive then
        startAutoFarmCycle()
    end
end
